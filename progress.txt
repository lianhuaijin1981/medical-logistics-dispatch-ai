================================================================================
  项目运行
================================================================================

# 安装依赖
pnpm install

# 启动后端 (Mock 模式 — 不需要数据库, 端口 3001)
cd apps/api && PORT=3001 pnpm dev:mock:once

# 启动后端 (真实模式 — 需要 MongoDB, 端口 3001)
cd apps/api && PORT=3001 pnpm dev

# 启动前端 (端口 3000)
cd apps/web && pnpm dev

# 全量构建
pnpm run build

# Swagger 文档
http://localhost:3001/api/docs

# MongoDB Shell
mongosh medical_logistics
